ggplot()+ geom_line(data = wage_by_edu, aes(x = year, y = less_than_hs,color ="Less High")) + 
  geom_line(data= wage_by_edu,aes(x = year, y = high_school, color="High School"))+

  geom_line(data= wage_by_edu,aes(x = year, y =advanced_degree,color ="Advanced Degree"))+ 
  labs(title ="Average Wages By Education level",x ="Years",y="Average wages")+
   scale_color_manual(values = c(
     "Less High" = 'black',
      "High School" = 'red',
      "Some College" = 'blue',
      "Bachelor's Degree" = 'green',
      "Advanced Degree"='orange'
    )
  ) +theme_minimal()
