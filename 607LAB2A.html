Once the survey responses have been collected, I will store the data in a PostgreSQL database. ratings_df<- dbGetQuery(con,
"SELECT
  u.users_name,
  m.movie_title,
  r.rating
FROM ratings r
JOIN users u
  ON r.user_id = u.users_id
JOIN movies m
  ON r.movies_id = m.movies_id
ORDER BY u.users_name, m.movie_title;")


ratings_df <- ratings_df %>%
 mutate(
  rating_status = ifelse(is.na(rating),
              "Not Seen",
              as.character(rating))
 )

movie_summary <- ratings_df %>%
